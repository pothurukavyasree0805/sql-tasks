SQL Practical Project: Customer Data Cleaning & Validation System.
Queries:

Create Database: 
create database project;
use project;

Create Customer Table:
Create table customers (
    customer_id int primary key,
    full_name varchar(100),
    email varchar (100),
    phone varchar (20),
    city varchar (50),
    username varchar (50),
    comments text);

SHOW VARIABLES LIKE 'local_infile';
SET GLOBAL local_infile = 1;

Import Raw Customer Data:
LOAD DATA LOCAL INFILE 'C:/Users/LENOVO/OneDrive/Desktop/D13/sql/customers.csv'
INTO TABLE customers
FIELDS TERMINATED BY ','
ENCLOSED BY '"'
LINES TERMINATED BY '\n'
IGNORE 1 ROWS;

View Table:
select * from customers; 

Create Duplicate Table:
create table customers_copy as
select * from customers; 

select *from customers_copy;

Convert to Lower Case:
update customers_copy
set full_name = lower(full_name),
     email = lower(email),
     city = lower(city),
     comments = lower(comments);

SET SQL_SAFE_UPDATES = 0;

Trim Extra Spaces:
update customers_copy
set full_name = trim(full_name),
    email = trim(email),
    city = trim(city),
    username = trim(username),
    comments = trim(comments);

Invalid Phone Number:
update customers_copy
set phone = replace(phone,'-','');
update customers_copy
set phone = replace(phone,' ','');    

Duplicate Records:
Select full_name, 
       email,
       phone,
       city,
       username,
       comments,
       count(*) as duplicate_count
from customers_copy
group by full_name,
         email,
         phone,
         city,
         username,
         comments
having count(*) > 1;

delete c1
from customers_copy c1
join customers_copy c2
on c1.email = c2.email
and c1.customer_id > c2.customer_id;

Incorrect Email Format:
update customers_copy
set email = concat(email, '@gmail.com')
where email not like '%@%';

Missing Value:
update customers_copy
set city = 'unknown'
where city is null or city = '';
update customers_copy
set comments = 'no comments'
where comments is null or comments = '';
update customers_copy
set phone = '0000000000'
where phone is null or phone = '';
update customers_copy
set email = 'unknown@gmail.com'
where email is null or email = '';
update customers_copy
set username = 'unknown_user'
where username is null or username = '';

Special Characters:
update customers_copy
set 
    full_name = regexp_replace(full_name, '[^a-za-z ]', ''),
    city = regexp_replace(city, '[^a-za-z ]', ''),
    phone = regexp_replace(phone, '[^0-9]', ''),
    username = regexp_replace(username, '[^a-za-z0-9]', '_');
   
Inconsistent City Names:
select city from customers_copy
group by city;

update customers_copy
set city = case
    when city in ('hyd', 'hyd.', 'hyderbad') then 'hyderabad'
    when city in ('chn', 'madras') then 'chennai'
    when city in ('bng', 'banglore', 'bengluru') then 'bangalore'
    when city in ('mum', 'bombay') then 'mumbai'
    when city in ('del', 'new delhi') then 'delhi'
    when city in ('puna') then 'pune'
    when city in ('kol', 'calcutta') then 'kolkata'
    when city in ('gova') then 'goa'
    else city
end;

Backup the Table:
